// Frozen round0 independent positive-series continuous-fill solution.
// Cp*theta_p'=-UA*(theta_p-theta_w); d(r*t*cw*theta_w)/dt=UA*(theta_p-theta_w).
function reference(elapsed:number,inlet:number){
 const Cp=.5*1840,r=.025/3,UA=24,cw=4186,M=.5/3,theta0=360-inlet,tfill=M/r,t=Math.min(Math.max(elapsed,0),tfill)
 if(t===0)return{Tp:360,Tw:inlet}
 const alpha=UA/Cp,beta=UA/(r*cw),z=alpha*t
 let sum=1,termExp=1
 for(let n=1;n<10000;n++){termExp*=z/n;const term=termExp*beta/(beta+n);sum+=term;if(Math.abs(term)<Math.abs(sum)*1e-15)break;if(n===9999)throw Error('Series did not converge')}
 const theta=theta0*Math.exp(-z)*sum
 let Tp=inlet+theta,Tw=inlet+(Cp*theta0-Cp*theta)/(r*t*cw)
 if(elapsed>tfill){const Cw=M*cw,Teq=inlet+Cp*theta0/(Cp+Cw),delta=(Tp-Tw)*Math.exp(-UA*(1/Cp+1/Cw)*(elapsed-tfill));Tp=Teq+Cw/(Cp+Cw)*delta;Tw=Teq-Cp/(Cp+Cw)*delta}
 return{Tp,Tw}
}
function rawBalance(label:string,c:PanelConfig,s:ReturnType<ReturnType<typeof createRun>['snapshot']>){
 const t=c.thermal,p=s.contact!.patches,l=s.contact!.ledger
 const initial=p.reduce((sum,q)=>sum+q.peatMassKg*1840*(t.initialK-t.inletK),0)
 const final=p.reduce((sum,q)=>sum+q.peatMassKg*1840*(q.temperatureK-t.inletK)+q.liquidWaterKg*4186*(q.waterTemperatureK-t.inletK)+q.waterVaporKg*(4186*(373.15-t.inletK)+2260000),0)
 const mass=t.waterKg-l.waterRemainingSupplyKg-l.waterBypassKg-p.reduce((sum,q)=>sum+q.liquidWaterKg+q.waterVaporKg,0)
 near(label+' raw mass',mass,0,1e-9+1e-9*t.waterKg)
 near(label+' raw energy',initial-final,0,1e-4+1e-9*Math.abs(initial))
 check(label+' finite bounds',p.every(q=>Number.isFinite(q.temperatureK)&&Number.isFinite(q.waterTemperatureK)&&q.temperatureK>=t.inletK-1e-8&&q.temperatureK<=t.initialK+1e-8&&q.waterTemperatureK<=373.15+1e-8&&q.liquidWaterKg>=-1e-12&&q.waterVaporKg>=-1e-12),p.map(q=>({Tp:q.temperatureK,Tw:q.waterTemperatureK,liquid:q.liquidWaterKg,vapor:q.waterVaporKg})),'bounded and nonnegative')
 check(label+' no reaction state',s.metrics!.oxygenKg===null&&s.metrics!.fuelKg===null&&s.metrics!.reactionW===null&&!s.fem,s.metrics,'O2/fuel/reaction unavailable; no field')
 return{timeS:s.timeS,massResidualKg:mass,energyResidualJ:initial-final}
}
